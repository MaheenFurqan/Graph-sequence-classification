bool isWalk(string seq[], int len)
{
	if (len>=2)
	{
		string current_city_name, next_city_name;
		int current_city_index, next_city_index;
		bool edge_exist;
		for (int index=0; index<len; index++)
		{
			if (index<len-1)
			{
				current_city_name=seq[index];
				current_city_index=getCityIndex(current_city_name);
				next_city_name=seq[index+1];
				next_city_index=getCityIndex(next_city_name);

                if ((next_city_index==-1)||(current_city_index==-1)) return false;
				
				edge_exist=edgeExists(current_city_name, next_city_name);
				
				if (edge_exist==false)
				{
					return false;
				}
			}
			
		}
		return true;
	}
	else return false;
}
//implement and return appropriate values


bool isTrail(string seq[], int len) 
{
    if (isWalk(seq, len)==false)
    {
        return false;
    }
    int used_edges[N][N];
    for (int i=0; i<N; i++)
    {
        for (int j=0; j<N; j++)
        {
            used_edges[i][j]=0;
        }
    }
    
    string current_city_name, next_city_name;
    int current_city_index, next_city_index;
    for (int index=0; index<len-1; index++)
    {
        current_city_name=seq[index];
        current_city_index=getCityIndex(current_city_name);

        next_city_name=seq[index+1];
        next_city_index=getCityIndex(next_city_name);

        if ((next_city_index==-1)||(current_city_index==-1)) return false;

        if (edgeExists(current_city_name, next_city_name)==true)
        {
            used_edges[current_city_index][next_city_index]+=1;
            used_edges[next_city_index][current_city_index]+=1;
        }
    }
    for (int i=0; i<N; i++)
    {
        for (int j=0; j<N; j++)
        {
            if (used_edges[i][j]>1)
            {
                return false;
            }
        }
    }
    return true;
}//implement and return appropriate values


bool isPath(string seq[], int len) 
{ 
    if (isWalk(seq, len)==false)
    return false;
    int count;
    for (int i=0; i<len-1; i++)
    {
        count=0;
        for (int j=i+1; j<len; j++)
        {
            if (seq[i]==seq[j])
            {
                
                count+=1;
            }}
            if (count>0)
            return false;
    }
    return true;

 }//implement and return appropriate values


bool isClosedWalk(string seq[], int len) 
{
    if (isWalk(seq, len)==false)
    {
        return false;
    }
    else
    {
        if (seq[0]==seq[len-1])
        {
            return true;
        }
        else return false;
    }   
}//implement and return appropriate values


bool isCircuit(string seq[], int len) 
{ 
    if (isClosedWalk(seq, len)==true && (len-1)>=3 && isTrail(seq,len)==true)
    {
        return true;
    }
    else
    {
        return false;
    }

}//implement and return appropriate values


bool isSimpleCircuit(string seq[], int len) 
{
    if (isCircuit(seq, len)==false)
    {
        return false;
    }
    int count;
    for (int i=0; i<len-1; i++)
    {
        count=0;
        for (int j=i+1; j<len-1; j++)
        {
            if (seq[i]==seq[j])
            {
                count+=1;
            }}
            if (count>0)
            return false;
    }
   return true;
}//implement and return appropriate values

// ----- Utility Functions -----
int getCityIndex(string name){
    for(int i=0;i<N;i++){
        if(cities[i]==name) return i;
    }
    return -1;
}

string toUpper(string s){
    for(int i=0;i<s.size();i++) s[i] = toupper(s[i]);
    return s;
}

bool edgeExists(string a, string b){
    int u=getCityIndex(a);
    int v=getCityIndex(b);
    if(u==-1 || v==-1) return false;
    return graph[u][v]==1;
}

int parseSequence(string input, string seq[]){ 
    // Students implement this
    
    string city="";
    int i=0;
    for (char character: input)
    {
    	if (character!='-'&&character!='>')
    	{
    		city=city+character;
		}
		else if (character=='-')
		{
			seq[i]=toUpper(city);
			city="";
			i+=1;
		}
	}
	seq[i]=toUpper(city);
    return i+1; 
}

// ----- Map Display -----
void printMap(){
    cout<<"\n--- Map ---\n";
    for(int i=0;i<N;i++){
        cout<<cities[i]<<" connected to: ";
        bool first=true;
        for(int j=0;j<N;j++){
            if(graph[i][j]){
                if(!first) cout<<", ";
                cout<<cities[j];
                first=false;
            }
        }
        if(first) cout<<"None";
        cout<<"\n";
    }
}

// ----- Grading -----
int marks[6] = {0,0,0,0,0,0};
string moduleNames[6] = {"Path","Walk","Trail","Closed Walk","Circuit","Simple Circuit"};

void grade(string seq[], int len, bool expected[6]){
    bool results[6] = {
        isPath(seq,len),
        isWalk(seq,len),
        isTrail(seq,len),
        isClosedWalk(seq,len),
        isCircuit(seq,len),
        isSimpleCircuit(seq,len)
    };

    for(int i=0;i<6;i++){
        if(results[i] == expected[i]) marks[i] = 5;
        else marks[i] = 0;
    }
}
